# BEGIN Atardecer endurecimiento
# Sin listado de carpetas y sin acceso directo a archivos que no son para el público.
Options -Indexes
<FilesMatch "^(wp-config\.php|readme\.html|license\.txt|xmlrpc\.php|\.htaccess.*|.*\.(bak|bk|old|orig|sql|log|ini|sh|zip|tar|gz))$">
	Order allow,deny
	Deny from all
</FilesMatch>
# Las páginas no se guardan en el navegador (así cada visita ve la portada al día);
# las imágenes, estilos y scripts sí, según las reglas de abajo.
<IfModule mod_expires.c>
	ExpiresActive On
	ExpiresByType text/html "access plus 0 seconds"
</IfModule>
# END Atardecer endurecimiento
